# tk-menus | Menus and dialogs | Menu bars, right-click menus, message boxes, file pickers | tkinter

@@ tk.menubar
Q: How do I add a menu bar (File, Edit, Help)?
aka: menu bar, file menu, top menu, menubar, add_cascade, add_command, dropdown menu at top
A: Create a `tk.Menu`, add sub-menus with `add_cascade`, and attach it with `root.config(menu=menubar)`.
see: tk.menu-accelerator, tk.right-click-menu, tk.menu-check
```py
import tkinter as tk

root = tk.Tk()
root.geometry("320x180")
menubar = tk.Menu(root)

file_menu = tk.Menu(menubar, tearoff=False)
file_menu.add_command(label="New", command=lambda: print("new"))
file_menu.add_command(label="Open...", command=lambda: print("open"))
file_menu.add_separator()
file_menu.add_command(label="Exit", command=root.destroy)
menubar.add_cascade(label="File", menu=file_menu)

help_menu = tk.Menu(menubar, tearoff=False)
help_menu.add_command(label="About", command=lambda: print("about"))
menubar.add_cascade(label="Help", menu=help_menu)

root.config(menu=menubar)
root.mainloop()
```
`tearoff=False` removes the dashed line at the top of each menu. On macOS the menu appears in the screen's top bar, not in the window.

@@ tk.menu-accelerator
Q: How do I show keyboard shortcuts in a menu (and make them work)?
aka: menu shortcut, accelerator, ctrl+s in menu, menu hotkey, underline letter menu
A: `accelerator="Ctrl+S"` only **displays** the text. You must also `bind` the key yourself.
see: tk.shortcuts
```py
import tkinter as tk

root = tk.Tk()
def save(event=None):
    print("saved")

menubar = tk.Menu(root)
fm = tk.Menu(menubar, tearoff=False)
fm.add_command(label="Save", accelerator="Ctrl+S", command=save, underline=0)
menubar.add_cascade(label="File", menu=fm, underline=0)     # Alt+F opens it on Windows
root.config(menu=menubar)
root.bind("<Control-s>", save)
root.mainloop()
```

@@ tk.submenu
Q: How do I make a sub-menu inside a menu?
aka: nested menu, submenu, cascade inside menu, menu inside menu, recent files menu
A: Make another `tk.Menu` whose parent is the menu, and `add_cascade` it there.
```py
import tkinter as tk

root = tk.Tk()
bar = tk.Menu(root)
file_menu = tk.Menu(bar, tearoff=False)
recent = tk.Menu(file_menu, tearoff=False)
for f in ("notes.txt", "todo.txt"):
    recent.add_command(label=f, command=lambda f=f: print("open", f))
file_menu.add_cascade(label="Open recent", menu=recent)
bar.add_cascade(label="File", menu=file_menu)
root.config(menu=bar)
root.mainloop()
```

@@ tk.menu-check
Q: How do I add checkmarks or radio choices to a menu?
aka: checkbutton menu, add_checkbutton, add_radiobutton, menu toggle, view menu options
A: `add_checkbutton(label=..., variable=bool_var)` and `add_radiobutton(label=..., variable=var, value=...)`.
```py
import tkinter as tk

root = tk.Tk()
wrap = tk.BooleanVar(value=True)
size = tk.StringVar(value="M")
bar = tk.Menu(root)
view = tk.Menu(bar, tearoff=False)
view.add_checkbutton(label="Word wrap", variable=wrap, command=lambda: print("wrap", wrap.get()))
view.add_separator()
for s in "SML":
    view.add_radiobutton(label=f"Size {s}", variable=size, value=s)
bar.add_cascade(label="View", menu=view)
root.config(menu=bar)
root.mainloop()
```

@@ tk.menu-disable
Q: How do I disable or change a menu item?
aka: grey out menu item, entryconfig, disable menu option, rename menu item
A: `menu.entryconfig("Label text", state="disabled")` (or use its index). Change the label the same way: `entryconfig(0, label="New name")`.
```py
import tkinter as tk

root = tk.Tk()
bar = tk.Menu(root)
edit = tk.Menu(bar, tearoff=False)
edit.add_command(label="Undo")
edit.add_command(label="Paste")
edit.entryconfig("Undo", state="disabled")
edit.entryconfig(1, label="Paste here")
bar.add_cascade(label="Edit", menu=edit)
root.config(menu=bar)
root.mainloop()
```

@@ tk.right-click-menu
Q: How do I make a right-click (context) menu?
aka: context menu, popup menu, right click menu, tk_popup, post menu
A: Make a `tk.Menu` and show it with `menu.tk_popup(event.x_root, event.y_root)` in a right-click handler.
see: tk.mouse
```py
import tkinter as tk
import sys

root = tk.Tk()
t = tk.Text(root, width=40, height=6)
t.pack(padx=10, pady=10)
menu = tk.Menu(root, tearoff=False)
menu.add_command(label="Cut", command=lambda: t.event_generate("<<Cut>>"))
menu.add_command(label="Copy", command=lambda: t.event_generate("<<Copy>>"))
menu.add_command(label="Paste", command=lambda: t.event_generate("<<Paste>>"))
menu.add_separator()
menu.add_command(label="Select all", command=lambda: t.tag_add("sel", "1.0", "end"))

def popup(event):
    try:
        menu.tk_popup(event.x_root, event.y_root)
    finally:
        menu.grab_release()

t.bind("<Button-2>" if sys.platform == "darwin" else "<Button-3>", popup)
root.mainloop()
```

@@ tk.messagebox
Q: How do I show a popup message (info, warning, error)?
aka: messagebox, alert, popup message, showinfo, showerror, showwarning, message box, dialog box
A: `from tkinter import messagebox`, then `messagebox.showinfo("Title", "Message")`. Also `showwarning` and `showerror`.
see: tk.messagebox-ask
```py
import tkinter as tk
from tkinter import messagebox

root = tk.Tk()
tk.Button(root, text="Info", command=lambda: messagebox.showinfo("Saved", "Your file was saved.")).pack(padx=40, pady=4)
tk.Button(root, text="Warning", command=lambda: messagebox.showwarning("Careful", "Low battery")).pack(pady=4)
tk.Button(root, text="Error", command=lambda: messagebox.showerror("Oops", "Something went wrong")).pack(pady=4)
root.mainloop()
```
If you get `module 'tkinter' has no attribute 'messagebox'`, add `from tkinter import messagebox`. It is a separate module.

@@ tk.messagebox-ask
Q: How do I ask a yes/no question in a popup?
aka: askyesno, yes no dialog, confirm dialog, askokcancel, askquestion, askyesnocancel, are you sure
A: `messagebox.askyesno("Title", "Question?")` returns `True` or `False`. `askyesnocancel` returns `True`/`False`/`None`.
see: tk.close-protocol
```py
import tkinter as tk
from tkinter import messagebox

root = tk.Tk()

def delete():
    if messagebox.askyesno("Delete", "Delete all items?"):
        print("Deleted")
    else:
        print("Kept")

tk.Button(root, text="Delete everything", command=delete).pack(padx=40, pady=40)
root.mainloop()
```
Return values: `askokcancel`/`askyesno`/`askretrycancel` → `True`/`False`; `askquestion` → the strings `"yes"`/`"no"`.

@@ tk.messagebox-no-root
Q: How do I show a messagebox without an empty main window behind it?
aka: messagebox only, no root window, hide root, popup without window, messagebox from script
A: Create the root, hide it with `withdraw()`, then show the box.
```py
import tkinter as tk
from tkinter import messagebox

root = tk.Tk()
root.withdraw()
messagebox.showinfo("Done", "The script has finished.")
root.destroy()
```

@@ tk.filedialog-open
Q: How do I let the user choose a file to open?
aka: open file dialog, file picker, browse file, askopenfilename, choose file, select file
A: `from tkinter import filedialog`, then `path = filedialog.askopenfilename(filetypes=[...])`. It returns `""` if the user cancels.
see: tk.filedialog-save, py.file-read
```py
import tkinter as tk
from tkinter import filedialog

root = tk.Tk()
text = tk.Text(root, width=50, height=10)
text.pack(padx=10, pady=10)

def open_file():
    path = filedialog.askopenfilename(title="Open", filetypes=[("Text files", "*.txt"), ("All files", "*.*")])
    if not path:
        return                     # cancelled
    with open(path, encoding="utf-8") as f:
        text.delete("1.0", "end")
        text.insert("1.0", f.read())
    root.title(path)

tk.Button(root, text="Open...", command=open_file).pack(pady=(0, 10))
root.mainloop()
```
`askopenfilenames` (with an s) lets them pick several and returns a tuple.

@@ tk.filedialog-save
Q: How do I show a "Save as" dialog?
aka: save file dialog, asksaveasfilename, save as, choose where to save, export file
A: `filedialog.asksaveasfilename(defaultextension=".txt", filetypes=[...])`, then write to that path.
```py
import tkinter as tk
from tkinter import filedialog

root = tk.Tk()
text = tk.Text(root, width=40, height=6)
text.pack(padx=10, pady=10)

def save_as():
    path = filedialog.asksaveasfilename(defaultextension=".txt", initialfile="notes.txt",
                                        filetypes=[("Text files", "*.txt")])
    if path:
        with open(path, "w", encoding="utf-8") as f:
            f.write(text.get("1.0", "end-1c"))

tk.Button(root, text="Save as...", command=save_as).pack(pady=(0, 10))
root.mainloop()
```

@@ tk.filedialog-folder
Q: How do I let the user pick a folder?
aka: choose directory, askdirectory, select folder, browse folder
A: `filedialog.askdirectory()` returns the folder path, or `""` if cancelled.
see: py.listdir
```py
import tkinter as tk
from tkinter import filedialog
import os

root = tk.Tk()
lb = tk.Listbox(root, width=50)
lb.pack(padx=10, pady=10)

def pick():
    folder = filedialog.askdirectory(title="Pick a folder")
    if folder:
        lb.delete(0, "end")
        for name in sorted(os.listdir(folder)):
            lb.insert("end", name)

tk.Button(root, text="Choose folder", command=pick).pack(pady=(0, 10))
root.mainloop()
```

@@ tk.simpledialog
Q: How do I ask the user to type a value in a popup?
aka: input dialog, askstring, askinteger, prompt, simpledialog, input box popup, ask for name popup
A: `from tkinter import simpledialog`, then `simpledialog.askstring("Title", "Prompt")`. Also `askinteger` and `askfloat` (with `minvalue`/`maxvalue`). They return `None` if cancelled.
see: tk.modal
```py
import tkinter as tk
from tkinter import simpledialog

root = tk.Tk()
lab = tk.Label(root, text="Hello, stranger", padx=30, pady=20)
lab.pack()

def ask():
    name = simpledialog.askstring("Name", "What's your name?", parent=root)
    age = simpledialog.askinteger("Age", "How old are you?", minvalue=1, maxvalue=120, parent=root)
    if name:
        lab.config(text=f"Hello, {name} ({age})")

tk.Button(root, text="Ask", command=ask).pack(pady=(0, 10))
root.mainloop()
```

@@ tk.colorchooser
Q: How do I let the user pick a colour?
aka: color picker, colour picker, askcolor, colorchooser, choose color dialog
A: `from tkinter import colorchooser`, then `rgb, hex_code = colorchooser.askcolor()`. Both are `None` if cancelled.
```py
import tkinter as tk
from tkinter import colorchooser

root = tk.Tk()
sw = tk.Label(root, text="  ", width=20, height=3, bg="white")
sw.pack(padx=20, pady=10)

def pick():
    rgb, hex_code = colorchooser.askcolor(title="Pick a colour", initialcolor=sw["bg"])
    if hex_code:
        sw.config(bg=hex_code)

tk.Button(root, text="Choose colour", command=pick).pack(pady=(0, 10))
root.mainloop()
```

@@ tk.fontchooser
Q: Is there a font picker dialog?
aka: font dialog, choose font, font picker
A: Not in standard tkinter (Tk 8.6+ has one but Python doesn't wrap it nicely). Easiest: a Listbox of `tkinter.font.families()` with a preview label.
see: tk.font-list
```py
import tkinter as tk
from tkinter import font

root = tk.Tk()
lb = tk.Listbox(root, height=10, width=30)
lb.pack(side="left", fill="y", padx=6, pady=6)
for fam in sorted(set(font.families())):
    lb.insert("end", fam)
preview = tk.Label(root, text="The quick brown fox", width=24)
preview.pack(side="left", padx=10)
lb.bind("<<ListboxSelect>>", lambda e: lb.curselection() and preview.config(font=(lb.get(lb.curselection()[0]), 16)))
root.mainloop()
```
