# tk-entry-text | Text input | Entry boxes and multi-line Text boxes | tkinter

@@ tk.entry-get
Q: How do I get what the user typed in an Entry box?
aka: entry get, read input, get text from entry, textbox value, input box, user input gui, entry value
A: Call `entry.get()` **when it's needed** (e.g. in a button's function), not right after creating the Entry, when it's still empty.
see: tk.entry-number, tk.err.none-grid, tk.button-enter
```py
import tkinter as tk

root = tk.Tk()
name = tk.Entry(root, width=25)
name.pack(padx=20, pady=10)
out = tk.Label(root, text="")
out.pack()

def greet():
    out.config(text=f"Hello, {name.get()}!")

tk.Button(root, text="Greet", command=greet).pack(pady=10)
root.mainloop()
```
`get()` always returns a **string**. See [[tk.entry-number]] for numbers.

@@ tk.entry-number
Q: How do I get a number from an Entry (and handle bad input)?
aka: entry int, convert entry to integer, entry float, ValueError entry, invalid literal for int entry, number input tkinter, calculator input
A: Convert with `int()` or `float()` inside `try`/`except ValueError` so letters or an empty box don't crash the program.
see: tk.entry-validate, py.err.valueerror-int
```py
import tkinter as tk

root = tk.Tk()
a = tk.Entry(root, width=8); a.grid(row=0, column=0, padx=5, pady=10)
tk.Label(root, text="+").grid(row=0, column=1)
b = tk.Entry(root, width=8); b.grid(row=0, column=2, padx=5)
result = tk.Label(root, text="")
result.grid(row=1, column=0, columnspan=3)

def add():
    try:
        total = float(a.get()) + float(b.get())
    except ValueError:
        result.config(text="Please type two numbers", fg="red")
        return
    result.config(text=f"= {total:g}", fg="black")

tk.Button(root, text="Add", command=add).grid(row=2, column=0, columnspan=3, pady=10)
root.mainloop()
```

@@ tk.entry-set
Q: How do I put text into an Entry, or clear it?
aka: entry insert, set entry text, clear entry, delete entry text, reset input box, empty textbox, entry default value
A: Clear with `entry.delete(0, "end")`, insert with `entry.insert(0, "text")`. Or use a `StringVar`: `var.set("text")`.
see: tk.stringvar
```py
import tkinter as tk

root = tk.Tk()
e = tk.Entry(root)
e.pack(padx=20, pady=10)
e.insert(0, "default value")

tk.Button(root, text="Clear", command=lambda: e.delete(0, "end")).pack(side="left", padx=20, pady=10)
tk.Button(root, text="Fill", command=lambda: (e.delete(0, "end"), e.insert(0, "filled"))).pack(side="left", pady=10)
root.mainloop()
```
`"end"` (or `tk.END`) means "after the last character".

@@ tk.entry-password
Q: How do I make a password box that shows stars?
aka: password entry, hide input, show asterisks, mask text, show password checkbox
A: `tk.Entry(root, show="*")`. Set `show=""` to reveal it again.
see: tk.recipe-login
```py
import tkinter as tk

root = tk.Tk()
pw = tk.Entry(root, show="*")
pw.pack(padx=20, pady=10)
visible = tk.BooleanVar()

def toggle():
    pw.config(show="" if visible.get() else "*")

tk.Checkbutton(root, text="Show password", variable=visible, command=toggle).pack(pady=(0, 10))
root.mainloop()
```

@@ tk.entry-placeholder
Q: How do I add placeholder (hint) text to an Entry?
aka: placeholder, hint text, greyed text in entry, watermark text, default text that disappears, ghost text
A: tkinter has no placeholder option. Insert grey text, and remove it when the box gets focus (`<FocusIn>`); put it back on `<FocusOut>` if empty.
```py
import tkinter as tk

class HintEntry(tk.Entry):
    def __init__(self, master, hint, **kw):
        super().__init__(master, **kw)
        self.hint = hint
        self.normal = self["fg"]
        self.bind("<FocusIn>", self._in)
        self.bind("<FocusOut>", self._out)
        self._out()

    def _in(self, e=None):
        if self["fg"] == "grey":
            self.delete(0, "end")
            self.config(fg=self.normal)

    def _out(self, e=None):
        if not super().get():
            self.insert(0, self.hint)
            self.config(fg="grey")

    def get(self):
        return "" if self["fg"] == "grey" else super().get()

root = tk.Tk()
HintEntry(root, "Your name").pack(padx=20, pady=10)
HintEntry(root, "Your email").pack(padx=20, pady=(0, 10))
root.mainloop()
```

@@ tk.entry-validate
Q: How do I only allow numbers (or limit length) in an Entry?
aka: validate entry, numbers only, digits only, restrict input, max length entry, validatecommand, vcmd, character limit
A: Use `validate="key"` with a `validatecommand` that returns `True` to allow the change. `%P` passes what the text **would** become.
see: tk.entry-number
```py
import tkinter as tk

root = tk.Tk()

def only_digits(new_text):
    return new_text == "" or new_text.isdigit()

def max_five(new_text):
    return len(new_text) <= 5

vd = root.register(only_digits)
vl = root.register(max_five)
tk.Label(root, text="Digits only").pack()
tk.Entry(root, validate="key", validatecommand=(vd, "%P")).pack(padx=20, pady=(0, 10))
tk.Label(root, text="Max 5 characters").pack()
tk.Entry(root, validate="key", validatecommand=(vl, "%P")).pack(padx=20, pady=(0, 10))
root.mainloop()
```
For decimals use a test like: try `float(new_text)` and return `False` on `ValueError` (allow `""`, `"-"` and `"."` too, so the user can start typing).

@@ tk.entry-readonly
Q: How do I make an Entry read-only (can select and copy but not type)?
aka: readonly entry, non editable entry, disable typing, entry state readonly, lock entry
A: `entry.config(state="readonly")`. `"disabled"` also blocks selecting. Switch to `"normal"` before changing the text from code.
```py
import tkinter as tk

root = tk.Tk()
e = tk.Entry(root)
e.insert(0, "You can copy me")
e.config(state="readonly")
e.pack(padx=20, pady=20)
root.mainloop()
```

@@ tk.entry-focus
Q: How do I put the cursor in an Entry automatically (focus)?
aka: focus entry, autofocus, focus_set, cursor in textbox at start, move to next entry, tab order
A: Call `entry.focus()` (same as `focus_set()`). Tab moves to the next widget in the order they were created.
```py
import tkinter as tk

root = tk.Tk()
first = tk.Entry(root); first.pack(padx=20, pady=5)
second = tk.Entry(root); second.pack(padx=20, pady=5)
first.focus()
first.bind("<Return>", lambda e: second.focus())   # Enter jumps to the next box
root.mainloop()
```
Change the Tab order with `widget.lift()` or by creating widgets in the order you want.

@@ tk.entry-width
Q: How do I make an Entry wider or taller?
aka: entry width, entry height, bigger textbox, large input, entry font size
A: `width=` is in characters. Entry is always one line; it gets taller with a bigger `font`. For several lines, use a [[tk.text|Text]] widget.
```py
import tkinter as tk

root = tk.Tk()
tk.Entry(root, width=40).pack(padx=20, pady=5)
tk.Entry(root, font=("Arial", 18)).pack(padx=20, pady=5)
tk.Entry(root).pack(fill="x", padx=20, pady=5)
root.mainloop()
```

@@ tk.entry-changes
Q: How do I run code every time the Entry text changes?
aka: entry on change, text changed event, live search, trace entry, detect typing, onchange
A: Give the Entry a `StringVar` and use `var.trace_add("write", callback)`.
see: tk.trace
```py
import tkinter as tk

root = tk.Tk()
text = tk.StringVar()
tk.Entry(root, textvariable=text).pack(padx=20, pady=10)
count = tk.Label(root, text="0 characters")
count.pack(pady=(0, 10))

def changed(*args):
    count.config(text=f"{len(text.get())} characters")

text.trace_add("write", changed)
root.mainloop()
```

@@ tk.text
Q: How do I make a multi-line text box?
aka: text widget, textarea, multiline input, big text box, notes box, text editor box, tk.Text
A: Use `tk.Text(root, width=40, height=10)`. Unlike Entry, its positions are `"line.column"` strings: start is `"1.0"`, end is `"end"`.
see: tk.text-get, tk.text-insert, tk.text-scroll
```py
import tkinter as tk

root = tk.Tk()
box = tk.Text(root, width=40, height=8, wrap="word")
box.pack(padx=10, pady=10)
box.insert("1.0", "Type anything here.\nIt can have many lines.")
root.mainloop()
```
`wrap="word"` wraps at spaces; `"char"` (default) can split words; `"none"` doesn't wrap.

@@ tk.text-get
Q: How do I get all the text from a Text widget?
aka: text get, read text widget, get contents of textarea, text get 1.0 end, get text box value
A: `text.get("1.0", "end-1c")`. `"1.0"` = line 1, character 0. `"end-1c"` drops the newline that Tk always adds at the end.
```py
import tkinter as tk

root = tk.Tk()
box = tk.Text(root, width=30, height=5)
box.pack(padx=10, pady=10)
box.insert("1.0", "line one\nline two")

def show():
    content = box.get("1.0", "end-1c")
    print(repr(content))
    print("lines:", content.count("\n") + 1)

tk.Button(root, text="Get", command=show).pack(pady=(0, 10))
root.mainloop()
```

@@ tk.text-insert
Q: How do I add, replace or clear text in a Text widget?
aka: text insert, text delete, clear text widget, append text, replace text, write to text box, log window
A: `insert("end", "more\n")` appends, `delete("1.0", "end")` clears everything. Call `see("end")` to scroll to the newest line.
see: tk.text-readonly
```py
import tkinter as tk

root = tk.Tk()
log = tk.Text(root, width=40, height=6)
log.pack(padx=10, pady=10)
n = 0

def add_line():
    global n
    n += 1
    log.insert("end", f"Message {n}\n")
    log.see("end")

tk.Button(root, text="Add", command=add_line).pack(side="left", padx=10, pady=(0, 10))
tk.Button(root, text="Clear", command=lambda: log.delete("1.0", "end")).pack(side="left", pady=(0, 10))
root.mainloop()
```

@@ tk.text-readonly
Q: How do I make a Text widget read-only?
aka: read only text, non editable text box, disable text widget, output box, stop typing in text
A: `text.config(state="disabled")`. To change it from code, switch to `"normal"`, edit, then back to `"disabled"`.
```py
import tkinter as tk

root = tk.Tk()
out = tk.Text(root, width=36, height=5, state="disabled")
out.pack(padx=10, pady=10)

def write(msg):
    out.config(state="normal")
    out.insert("end", msg + "\n")
    out.config(state="disabled")
    out.see("end")

tk.Button(root, text="Write", command=lambda: write("Read-only output")).pack(pady=(0, 10))
root.mainloop()
```

@@ tk.text-scroll
Q: How do I add a scrollbar to a Text widget?
aka: scroll text, scrollbar text box, scrolledtext, scrolling textarea
A: Easiest: `from tkinter import scrolledtext` and use `scrolledtext.ScrolledText`. Or connect a `Scrollbar` by hand.
see: tk.scrollbar
```py
import tkinter as tk
from tkinter import scrolledtext

root = tk.Tk()
box = scrolledtext.ScrolledText(root, width=40, height=8, wrap="word")
box.pack(padx=10, pady=10, fill="both", expand=True)
box.insert("1.0", "\n".join(f"Line {i}" for i in range(1, 51)))
root.mainloop()
```

@@ tk.text-tags
Q: How do I colour or style part of the text in a Text widget?
aka: text tag, highlight text, colored text, bold part of text, tag_configure, syntax highlight, change color of word
A: Create a tag with `tag_configure("name", foreground=..., font=...)`, then use it when inserting or with `tag_add(name, start, end)`.
see: tk.text-search
```py
import tkinter as tk

root = tk.Tk()
t = tk.Text(root, width=40, height=6, font=("Courier", 12))
t.pack(padx=10, pady=10)
t.tag_configure("red", foreground="red")
t.tag_configure("big", font=("Courier", 16, "bold"))
t.tag_configure("hl", background="yellow")

t.insert("end", "Normal ")
t.insert("end", "red ", "red")
t.insert("end", "big\n", "big")
t.insert("end", "highlight this word")
t.tag_add("hl", "2.10", "2.14")    # line 2, characters 10 to 14
root.mainloop()
```

@@ tk.text-search
Q: How do I find and highlight a word in a Text widget?
aka: search text, find in text, find all occurrences, highlight search results, text search
A: Loop with `text.search(word, start, stopindex="end")`; it returns the index of the next match or `""`.
see: tk.text-tags
```py
import tkinter as tk

root = tk.Tk()
t = tk.Text(root, width=40, height=5)
t.pack(padx=10, pady=10)
t.insert("1.0", "the cat sat on the mat with the hat")
t.tag_configure("found", background="yellow")
q = tk.Entry(root); q.pack(side="left", padx=10, pady=(0, 10)); q.insert(0, "the")

def find():
    t.tag_remove("found", "1.0", "end")
    word = q.get()
    if not word:
        return
    pos = "1.0"
    while True:
        pos = t.search(word, pos, stopindex="end", nocase=True)
        if not pos:
            break
        end = f"{pos}+{len(word)}c"
        t.tag_add("found", pos, end)
        pos = end

tk.Button(root, text="Find", command=find).pack(side="left", pady=(0, 10))
root.mainloop()
```

@@ tk.text-undo
Q: How do I add undo/redo to a Text widget?
aka: undo text, ctrl z text widget, redo, edit_undo
A: Create it with `undo=True`. Ctrl+Z / Ctrl+Y (Cmd+Z / Cmd+Shift+Z on Mac) then work, and you can call `edit_undo()` / `edit_redo()`.
```py
import tkinter as tk

root = tk.Tk()
t = tk.Text(root, width=40, height=6, undo=True, maxundo=-1)
t.pack(padx=10, pady=10)

def undo():
    try:
        t.edit_undo()
    except tk.TclError:
        pass      # nothing to undo

tk.Button(root, text="Undo", command=undo).pack(pady=(0, 10))
root.mainloop()
```

@@ tk.text-cursor
Q: How do I get or move the cursor position in a Text or Entry?
aka: insert cursor, cursor position, caret, current line number, move cursor end, index insert
A: The cursor is the `"insert"` index. Read it with `text.index("insert")`; move it with `text.mark_set("insert", "1.0")`. For an Entry use `entry.index("insert")` and `entry.icursor(n)`.
```py
import tkinter as tk

root = tk.Tk()
t = tk.Text(root, width=30, height=4)
t.pack(padx=10, pady=10)
t.insert("1.0", "hello\nworld")
pos = tk.Label(root)
pos.pack()

def update(e=None):
    line, col = t.index("insert").split(".")
    pos.config(text=f"Line {line}, column {int(col) + 1}")

t.bind("<KeyRelease>", update)
t.bind("<ButtonRelease-1>", update)
t.mark_set("insert", "end")
update()
root.mainloop()
```

@@ tk.text-indexes
Q: What do Text indexes like "1.0", "end-1c", "insert lineend" mean?
aka: text index, 1.0, end-1c, linestart, lineend, wordstart, text positions
A: `"line.column"`: lines start at 1, columns at 0. You can add modifiers: `+5c` (characters), `-1l` (lines), `linestart`, `lineend`, `wordstart`, `wordend`.
```py
import tkinter as tk

root = tk.Tk()
t = tk.Text(root)
t.insert("1.0", "first line\nsecond line\n")
print(t.get("1.0", "1.end"))              # first line
print(t.get("2.0", "2.6"))                # second
print(t.get("1.0", "1.0 + 5c"))           # first
print(t.get("2.3 wordstart", "2.3 wordend"))   # second
print(t.index("end-1c"))                  # 3.0 (very last position)
root.destroy()
```
