# tk-geo | Layout in depth | Every pack, grid and place option, propagation, uniform columns, responsive tricks | tkinter

@@ tk.geo.pack-options
Q: What are all the pack() options (side, fill, expand, anchor, padx, ipadx, before, after)?
aka: pack options, pack side, ipadx ipady, pack before after, pack in_, pack anchor, internal padding
A: `side` (`"top"` default, `"bottom"`, `"left"`, `"right"`), `fill` (`"x"`, `"y"`, `"both"`), `expand` (take spare space), `anchor` (where it sits in its space: `"n"`, `"ne"`, ...), `padx/pady` (space outside; a pair `(left, right)` is allowed), `ipadx/ipady` (space inside, makes the widget bigger), `before=`/`after=` (pack next to another widget), `in_=` (pack into a different parent).
see: tk.pack, tk.pack-fill
```py
import tkinter as tk

root = tk.Tk()
root.geometry("360x240")
tk.Label(root, text="top, fill x", bg="#ffd27a").pack(side="top", fill="x")
tk.Label(root, text="bottom, ipady 10", bg="#9bd").pack(side="bottom", fill="x", ipady=10)
tk.Label(root, text="left", bg="#e6a").pack(side="left", fill="y", padx=(10, 0))
tk.Label(root, text="expand + anchor se", bg="#cfc").pack(side="left", expand=True, anchor="se", padx=10, pady=10)
first = tk.Label(root, text="I was packed last, but before 'left'", bg="#eee")
first.pack(side="left", before=root.pack_slaves()[2])
print([w.cget("text") for w in root.pack_slaves()])
root.mainloop()
```

@@ tk.geo.pack-order
Q: Why does pack put my widgets in strange places? How does packing order work?
aka: pack order, pack confusing, pack left then top, widgets wrong place pack, cavity
A: Each `pack` call takes a whole strip from the **remaining** space (the "cavity") on its side. Once something is packed on the left, a later `side="top"` widget only gets the space to the right of it. Pack the widgets that should span the full width/height **first**, or group widgets in Frames.
see: tk.pack, tk.frame
```py
import tkinter as tk

root = tk.Tk()
root.geometry("300x160")
# toolbar and status bar first: they span the whole width
tk.Label(root, text="toolbar", bg="#ffd27a").pack(side="top", fill="x")
tk.Label(root, text="status bar", bg="#ddd").pack(side="bottom", fill="x")
tk.Label(root, text="side\npanel", bg="#9bd").pack(side="left", fill="y")
tk.Text(root).pack(side="left", fill="both", expand=True)       # the rest
root.mainloop()
```

@@ tk.geo.grid-options
Q: What are all the grid() options (row, column, sticky, rowspan, padx, ipadx, in_)?
aka: grid options list, grid ipadx, grid in_, grid_remove, grid_forget, grid_info, grid padding tuple
A: `row`, `column`, `rowspan`, `columnspan`, `sticky` (any of `n s e w`, e.g. `"nsew"`, `"ew"`), `padx/pady` (outside, can be a pair), `ipadx/ipady` (inside), `in_`. Related: `grid_remove()` hides but remembers options; `grid_forget()` forgets them; `grid_info()` reads them.
see: tk.grid, tk.grid-sticky, tk.grid-span
```py
import tkinter as tk

root = tk.Tk()
for r in range(3):
    for c in range(3):
        tk.Label(root, text=f"{r},{c}", bg="#eee", relief="ridge").grid(row=r, column=c, ipadx=10, ipady=6, padx=(0, 2), pady=(0, 2))
big = tk.Label(root, text="span 3 columns, sticky ew", bg="#ffd27a")
big.grid(row=3, column=0, columnspan=3, sticky="ew")
print(big.grid_info()["columnspan"], big.grid_info()["sticky"])
big.grid_remove()          # hidden...
big.grid()                 # ...and back in the same place
root.mainloop()
```

@@ tk.geo.columnconfigure
Q: What do columnconfigure/rowconfigure options do (weight, minsize, pad, uniform)?
aka: columnconfigure options, rowconfigure, weight, minsize, uniform, equal width columns, pad column
A: `weight` = how much extra space this column gets (0 = none); `minsize` = minimum pixels; `pad` = extra space added to the largest widget; `uniform="group"` = all columns with the same group name get the **same width** (in proportion to their weights).
see: tk.grid-weight
```py
import tkinter as tk

root = tk.Tk()
root.geometry("420x120")
for c, (w, label) in enumerate([(1, "weight 1"), (2, "weight 2"), (1, "weight 1")]):
    root.columnconfigure(c, weight=w, uniform="cols", minsize=60)
    tk.Label(root, text=label, bg=["#ffd27a", "#9bd", "#e6a"][c]).grid(row=0, column=c, sticky="nsew")
root.rowconfigure(0, weight=1)
root.update_idletasks()
print([root.grid_bbox(c, 0)[2] for c in range(3)])     # widths 1 : 2 : 1
root.mainloop()
```
```text
[105, 210, 105]
```

@@ tk.geo.place-options
Q: What are all the place() options (x, y, relx, rely, relwidth, anchor, bordermode)?
aka: place options, relx rely, relwidth relheight, place anchor, place in_, overlay widget, absolute position
A: `x`, `y` (pixels), `relx`, `rely` (0–1 of the parent's size), `width`, `height`, `relwidth`, `relheight`, `anchor` (which point of the widget sits at x/y), `in_`. Absolute and relative values add together: `relx=1, x=-10` = 10 px from the right edge.
see: tk.place, tk.center-widget
```py
import tkinter as tk

root = tk.Tk()
root.geometry("320x200")
tk.Label(root, text="centre", bg="#ffd27a").place(relx=0.5, rely=0.5, anchor="center")
tk.Label(root, text="10 px from bottom-right", bg="#9bd").place(relx=1, rely=1, x=-10, y=-10, anchor="se")
tk.Label(root, text="top half", bg="#eee").place(relx=0, rely=0, relwidth=1, relheight=0.4)
badge = tk.Label(root, text="3", bg="#c23b22", fg="white", font=("Arial", 9, "bold"))
btn = tk.Button(root, text="Inbox"); btn.place(x=20, y=120)
badge.place(in_=btn, relx=1, x=-4, y=-6, anchor="ne")      # badge on the button's corner
root.mainloop()
```

@@ tk.geo.propagate
Q: What does pack_propagate(False) / grid_propagate(False) do?
aka: pack_propagate, grid_propagate, frame shrinks, fixed size frame, propagate false
A: Normally a container shrinks or grows to fit its children. `frame.pack_propagate(False)` (or `grid_propagate(False)` if its children use grid) makes it keep the `width`/`height` you gave it.
see: tk.frame-size
```py
import tkinter as tk

root = tk.Tk()
fixed = tk.Frame(root, width=200, height=100, bg="#ffd27a")
fixed.pack_propagate(False)
fixed.pack(padx=10, pady=10)
tk.Button(fixed, text="small button").pack()

shrunk = tk.Frame(root, width=200, height=100, bg="#9bd")
shrunk.pack(padx=10, pady=10)
tk.Button(shrunk, text="small button").pack()
root.update_idletasks()
print(fixed.winfo_height(), shrunk.winfo_height() < 100)
root.mainloop()
```

@@ tk.geo.mix-managers
Q: Can I use pack in one frame and grid in another?
aka: mix pack and grid, pack and grid together, different frames different managers, combine layouts
A: Yes, as long as each **container** uses only one manager for its direct children. A common layout: the root uses `pack` for big areas; a form Frame inside uses `grid`.
see: tk.err.pack-grid
```py
import tkinter as tk

root = tk.Tk()
tk.Label(root, text="Header (pack)", bg="#ffd27a").pack(fill="x")
form = tk.Frame(root, padx=10, pady=10)
form.pack(fill="both", expand=True)                     # root uses pack
for r, label in enumerate(("Name", "Email")):           # form uses grid
    tk.Label(form, text=label).grid(row=r, column=0, sticky="w")
    tk.Entry(form).grid(row=r, column=1, sticky="ew")
form.columnconfigure(1, weight=1)
tk.Button(root, text="Save").pack(pady=(0, 10))
root.mainloop()
```

@@ tk.geo.sidebar
Q: How do I make a layout with a sidebar, a header and a main area?
aka: sidebar layout, app layout header sidebar content, navigation sidebar, master detail layout, admin layout
A: Use grid on the root: header row spanning both columns, sidebar in column 0 (fixed width), main area in column 1 with `weight=1` so it takes the rest.
see: tk.pages, tk.geo.columnconfigure
```py
import tkinter as tk

root = tk.Tk()
root.geometry("520x320")
root.columnconfigure(1, weight=1)
root.rowconfigure(1, weight=1)
tk.Label(root, text="My App", font=("Arial", 14, "bold"), bg="#1b1a17", fg="white", anchor="w", padx=12).grid(row=0, column=0, columnspan=2, sticky="ew")
side = tk.Frame(root, bg="#ece6da", width=140)
side.grid(row=1, column=0, sticky="ns")
main = tk.Frame(root, bg="white")
main.grid(row=1, column=1, sticky="nsew")
content = tk.Label(main, text="Home", bg="white", font=("Arial", 18))
content.pack(expand=True)
for name in ("Home", "Reports", "Settings"):
    tk.Button(side, text=name, relief="flat", bg="#ece6da", anchor="w", width=14,
              command=lambda n=name: content.config(text=n)).pack(fill="x", padx=6, pady=2)
root.mainloop()
```

@@ tk.geo.toolbar
Q: How do I make a toolbar (a row of icon buttons) at the top of the window?
aka: toolbar, icon toolbar, button bar top, toolbar with icons, ribbon
A: A Frame packed at the top with `fill="x"`, containing small flat Buttons packed `side="left"` (and a separator). Put it before the main content.
see: tk.geo.pack-order, tk.button-image
```py
import tkinter as tk

root = tk.Tk()
root.geometry("360x200")
bar = tk.Frame(root, bd=1, relief="raised")
bar.pack(side="top", fill="x")
icons = {}
for name, col in (("New", "#3b6ea8"), ("Open", "#e8b84a"), ("Save", "#2e7d4f")):
    img = tk.PhotoImage(width=16, height=16); img.put(col, to=(2, 2, 14, 14)); icons[name] = img
    tk.Button(bar, image=img, text=name, compound="left", relief="flat", padx=4,
              command=lambda n=name: print(n)).pack(side="left", padx=2, pady=2)
tk.Frame(bar, width=1, bg="#999").pack(side="left", fill="y", padx=4, pady=4)
tk.Button(bar, text="Help", relief="flat").pack(side="right", padx=2)
tk.Text(root).pack(fill="both", expand=True)
root.icons = icons
root.mainloop()
```

@@ tk.geo.flow-wrap
Q: How do I make widgets wrap onto the next row when the window is narrow (flow layout)?
aka: flow layout, wrap widgets, responsive grid of buttons, tags layout wrap, flexbox tkinter
A: Tk has no flow layout. On `<Configure>`, work out how many columns fit (`width // item_width`) and re-grid the items in rows of that many.
see: tk.responsive
```py
import tkinter as tk

root = tk.Tk()
root.geometry("360x200")
box = tk.Frame(root); box.pack(fill="both", expand=True, padx=6, pady=6)
chips = [tk.Label(box, text=t, bg="#ece6da", padx=8, pady=3) for t in
         ("python", "tkinter", "layout", "grid", "pack", "place", "canvas", "events", "threads", "ttk", "fonts", "images")]
ITEM_W = 80

def reflow(event=None):
    cols = max(1, box.winfo_width() // ITEM_W)
    for i, chip in enumerate(chips):
        chip.grid(row=i // cols, column=i % cols, padx=3, pady=3, sticky="w")

box.bind("<Configure>", reflow)
root.mainloop()
```

@@ tk.geo.aspect-frame
Q: How do I keep a widget square (or a fixed ratio) as the window resizes?
aka: keep square, fixed aspect ratio widget, square canvas resize, maintain ratio, board stays square
A: Place the widget inside a container and on the container's `<Configure>`, compute the biggest square that fits and `place` the widget with that size, centred.
see: tk.geo.place-options, tk.fullscreen-game-scale
```py
import tkinter as tk

root = tk.Tk()
root.geometry("400x260")
holder = tk.Frame(root, bg="#333"); holder.pack(fill="both", expand=True)
board = tk.Canvas(holder, bg="#f2ede3", highlightthickness=0)

def fit(e):
    side = min(e.width, e.height) - 10
    board.place(x=(e.width - side) // 2, y=(e.height - side) // 2, width=side, height=side)
    board.delete("all")
    for i in range(1, 3):
        board.create_line(side * i / 3, 0, side * i / 3, side, width=3)
        board.create_line(0, side * i / 3, side, side * i / 3, width=3)

holder.bind("<Configure>", fit)
root.mainloop()
```

@@ tk.geo.overlay
Q: How do I put a widget on top of another (overlay, floating button, badge)?
aka: overlay widget, floating button, widget on top of canvas, badge on icon, stack widgets on top
A: Use `place` for the top widget: it can sit anywhere over widgets managed by pack or grid. Use `lift()` if it ends up underneath.
see: tk.geo.place-options, tk.win.lift-lower
```py
import tkinter as tk

root = tk.Tk()
root.geometry("320x220")
tk.Text(root).pack(fill="both", expand=True)
fab = tk.Button(root, text="+", font=("Arial", 16, "bold"), bg="#c23b22", fg="white", relief="flat", width=3)
fab.place(relx=1, rely=1, x=-16, y=-16, anchor="se")
fab.lift()
root.mainloop()
```

@@ tk.geo.collapsible
Q: How do I make a collapsible section (click a header to show/hide its contents)?
aka: collapsible frame, accordion, expand collapse section, toggle panel, show more
A: A header button toggles the content Frame with `pack()`/`pack_forget()` (use `pack(after=header)` so it reappears in the right place).
see: tk.forget
```py
import tkinter as tk

class Section(tk.Frame):
    def __init__(self, master, title):
        super().__init__(master)
        self.open = tk.BooleanVar(value=True)
        self.head = tk.Button(self, text="▾ " + title, anchor="w", relief="flat", command=self.toggle)
        self.head.pack(fill="x")
        self.body = tk.Frame(self, padx=16)
        self.body.pack(fill="x")
        self.title = title

    def toggle(self):
        self.open.set(not self.open.get())
        if self.open.get():
            self.body.pack(fill="x", after=self.head)
        else:
            self.body.pack_forget()
        self.head.config(text=("▾ " if self.open.get() else "▸ ") + self.title)

root = tk.Tk()
for name in ("General", "Advanced"):
    s = Section(root, name); s.pack(fill="x", padx=8, pady=2)
    for i in range(3):
        tk.Checkbutton(s.body, text=f"{name} option {i + 1}").pack(anchor="w")
s.toggle()
root.mainloop()
```

@@ tk.geo.slaves
Q: How do I list the widgets inside a container, or the ones in a particular grid row?
aka: pack_slaves, grid_slaves, place_slaves, children of frame, widgets in row, get all widgets
A: `container.winfo_children()` = all child widgets; `pack_slaves()` / `place_slaves()` = those managed by that manager; `grid_slaves(row=r, column=c)` = widgets in that cell, row or column.
see: tk.grid-info
```py
import tkinter as tk

root = tk.Tk()
for r in range(3):
    for c in range(2):
        tk.Label(root, text=f"r{r}c{c}").grid(row=r, column=c)
print([w.cget("text") for w in root.grid_slaves(row=1)])
print(root.grid_size())                  # (columns, rows)
for w in root.grid_slaves(column=1):
    w.grid_forget()                      # remove a whole column
print(len(root.winfo_children()), len(root.grid_slaves()))
root.mainloop()
```
```text
['r1c1', 'r1c0']
(2, 3)
6 3
```

@@ tk.geo.padding-everywhere
Q: How do I add the same padding around every widget without typing padx/pady each time?
aka: default padding, padding all widgets, grid padding loop, consistent spacing, winfo_children configure padding
A: After creating them, loop over `frame.winfo_children()` and call `grid_configure(padx=..., pady=...)` (or `pack_configure`). ttk Frames also accept `padding=`.
see: tk.padding
```py
import tkinter as tk
from tkinter import ttk

root = tk.Tk()
form = ttk.Frame(root, padding=12); form.pack()
for r, label in enumerate(("Name", "Town", "Phone")):
    ttk.Label(form, text=label).grid(row=r, column=0, sticky="w")
    ttk.Entry(form).grid(row=r, column=1)
for w in form.winfo_children():
    w.grid_configure(padx=6, pady=4)
root.mainloop()
```

@@ tk.geo.centre-group
Q: How do I centre a group of widgets in the middle of the window (and keep them centred)?
aka: center group widgets, centre form in window, middle of window, center frame
A: Put them in a Frame, then either `frame.place(relx=0.5, rely=0.5, anchor="center")`, or grid the frame in a cell of a 1×1 grid with weight and no sticky (it stays centred).
see: tk.center-widget
```py
import tkinter as tk

root = tk.Tk()
root.geometry("400x260")
root.rowconfigure(0, weight=1); root.columnconfigure(0, weight=1)
card = tk.Frame(root, bd=1, relief="solid", padx=20, pady=16)
card.grid(row=0, column=0)            # no sticky: stays in the centre
tk.Label(card, text="Log in", font=("Arial", 16)).pack()
tk.Entry(card).pack(pady=4)
tk.Entry(card, show="•").pack(pady=4)
tk.Button(card, text="Go").pack(pady=(6, 0))
root.mainloop()
```

@@ tk.geo.scroll-both
Q: How do I make a canvas or frame that scrolls both horizontally and vertically?
aka: scroll horizontal and vertical, two scrollbars, xscrollcommand yscrollcommand, big area scroll both ways
A: Two scrollbars, both connected (`xscrollcommand` + `xview`, `yscrollcommand` + `yview`), laid out with grid so the corner works out, and a `scrollregion` covering everything.
see: tk.canvas-scroll, ttk.treeview-scroll
```py
import tkinter as tk

root = tk.Tk()
root.geometry("320x220")
c = tk.Canvas(root, bg="white", scrollregion=(0, 0, 1200, 900))
xs = tk.Scrollbar(root, orient="horizontal", command=c.xview)
ys = tk.Scrollbar(root, orient="vertical", command=c.yview)
c.config(xscrollcommand=xs.set, yscrollcommand=ys.set)
c.grid(row=0, column=0, sticky="nsew"); ys.grid(row=0, column=1, sticky="ns"); xs.grid(row=1, column=0, sticky="ew")
root.rowconfigure(0, weight=1); root.columnconfigure(0, weight=1)
for x in range(0, 1200, 100):
    for y in range(0, 900, 100):
        c.create_text(x + 50, y + 50, text=f"{x},{y}")
root.mainloop()
```

@@ tk.geo.debug-layout
Q: How can I see where my frames are while fixing a layout?
aka: debug layout, see frame borders, layout not working, show widget boundaries, coloured frames debug
A: Temporarily give every Frame a bright `bg` and a border (`highlightthickness=1, highlightbackground="red"`). A small function can do it for the whole window.
see: tk.frame-size
```py
import tkinter as tk

def show_frames(widget, depth=0):
    colours = ["#ffd6d6", "#d6ffd6", "#d6d6ff", "#ffffd6"]
    for w in widget.winfo_children():
        if isinstance(w, tk.Frame):
            w.config(bg=colours[depth % 4], highlightthickness=1, highlightbackground="red")
        show_frames(w, depth + 1)

root = tk.Tk()
outer = tk.Frame(root, padx=10, pady=10); outer.pack(fill="both", expand=True)
inner = tk.Frame(outer, padx=10, pady=10); inner.pack(side="left")
tk.Label(inner, text="Where am I?").pack()
show_frames(root)
root.mainloop()
```
